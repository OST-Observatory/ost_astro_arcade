% The feedback button as it looks on screen: a speech bubble with "!".
\newcommand{\fbicon}[1][0.36]{%
  \tikz[baseline=-0.6ex,scale=#1]{
    \fill[bg,rounded corners=1.2pt] (-0.62,-0.62) rectangle (0.62,0.62);
    \fill[white,rounded corners=0.8pt] (-0.36,-0.2) rectangle (0.36,0.38);

    \fill[bg] (-0.04,-0.03) rectangle (0.04,0.27);
    \fill[bg] (0,-0.12) circle (0.045);
  }}

% Numbered step: orange disc, text beside it.
\newcommand{\step}[2]{%
  \begin{tikzpicture}[baseline=(n.base)]
    \node[circle,fill=accent,text=bg,font=\bfseries\large,minimum size=8.5mm,inner sep=0] (n) {#1};
  \end{tikzpicture}\hspace{3.5mm}%
  \parbox[t]{108mm}{#2}\par\vspace{6mm}}

  \color{ink}\large
  \step{1}{Tap \fbicon\ at the \textbf{right edge} of the screen, right where you noticed it.
    \textcolor{dim}{\normalsize A screenshot of that moment is sent along.}}
  \step{2}{Choose \textbf{Bug}, \textbf{Idea} or \textbf{Other}.}
  \step{3}{If you like, type one short sentence on the on-screen keyboard.}
  \step{4}{Tap \textbf{Send}. \textcolor{accentdark}{\textbf{Thank you!}}}
\end{minipage}
